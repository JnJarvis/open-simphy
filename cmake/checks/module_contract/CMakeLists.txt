cmake_minimum_required(VERSION 3.28)
project(ModuleContract LANGUAGES NONE)
include("${MODULE_HELPERS}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp" "#pragma once\n")
if(CASE STREQUAL "valid")
    opensim_add_module(core HEADERS "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp")
    opensim_add_module(scene HEADERS "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp" PUBLIC_DEPS core)
elseif(CASE STREQUAL "forbidden")
    opensim_add_module(core HEADERS "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp" PUBLIC_DEPS math)
elseif(CASE STREQUAL "unknown")
    opensim_add_module(unknown HEADERS "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp")
elseif(CASE STREQUAL "missing")
    opensim_add_module(scene HEADERS "${CMAKE_CURRENT_BINARY_DIR}/include/probe.hpp" PUBLIC_DEPS core)
endif()
